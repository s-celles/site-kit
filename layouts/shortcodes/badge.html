{{- /* {{< badge "PASS" >}} or {{< badge "status" "proved_in_model" >}}: a status word in a badge (SPEC.md §5). Its tone (colour, border style) comes from the site parameter siteKit.badgeTones, a map from the word to one of the tones of site-kit.css; a word without a tone is neutral. Its label comes from siteKit.badgeLabels when the word has one, otherwise it is the word with underscores as spaces. The word is always shown, so no meaning rests on colour alone; data-status keeps the raw word. */ -}} {{- $kind := "" -}}{{- $word := .Get 0 -}} {{- with .Get 1 }}{{ $kind = $word }}{{ $word = . }}{{ end -}} {{- $tones := site.Params.siteKit.badgeTones | default dict -}} {{- $labels := site.Params.siteKit.badgeLabels | default dict -}} {{- $tone := index $tones (lower $word) | default "neutral" -}} {{- $label := index $labels (lower $word) | default (replace $word "_" " ") -}} {{ $label }}
